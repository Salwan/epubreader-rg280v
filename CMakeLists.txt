cmake_minimum_required(VERSION 3.20)
project(epubreader VERSION 0.5.8 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EPUBREADER_DEVICE "Build for OpenDingux" OFF)
option(EPUBREADER_SANITIZE "Host address/undefined sanitizers" OFF)
add_compile_options("-ffile-prefix-map=${PROJECT_SOURCE_DIR}=.")
if(EPUBREADER_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(PLATFORM REQUIRED IMPORTED_TARGET sdl SDL_ttf libmpg123 alsa)
find_package(Threads REQUIRED)
include(cmake/CRengine.cmake)
add_library(diagnostic_core
    src/ui/canvas.cpp src/platform/input.cpp src/platform/events.cpp src/storage/state.cpp
    src/platform/mixer.cpp src/platform/power.cpp src/audio/audio.cpp src/audio/catalog.cpp src/audio/playlist.cpp src/audio/music.cpp)
target_include_directories(diagnostic_core PUBLIC src)
target_link_libraries(diagnostic_core PUBLIC PkgConfig::PLATFORM Threads::Threads)
target_compile_options(diagnostic_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(production_storage src/storage/store.cpp src/storage/position.cpp)
target_include_directories(production_storage SYSTEM PUBLIC "${PROJECT_SOURCE_DIR}/third_party/src/json/single_include")
target_link_libraries(production_storage PUBLIC diagnostic_core)
target_compile_options(production_storage PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(library_core src/library/identity.cpp src/library/scanner.cpp src/library/catalog.cpp)
target_link_libraries(library_core PUBLIC reader_hash production_storage)
target_compile_options(library_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(reader_core src/reader/engine.cpp src/reader/metadata.cpp src/reader/worker.cpp src/ui/reader_controls.cpp)
target_link_libraries(reader_core PUBLIC crengine library_core)
target_compile_options(reader_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(application_core src/app/controller.cpp src/app/power_policy.cpp)
target_link_libraries(application_core PUBLIC reader_core)
target_compile_options(application_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(epubreader-diagnostic src/app/main.cpp src/ui/diagnostic.cpp)
target_link_libraries(epubreader-diagnostic PRIVATE diagnostic_core)
target_compile_options(epubreader-diagnostic PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_compile_definitions(epubreader-diagnostic PRIVATE EPUBREADER_VERSION="${PROJECT_VERSION}")
add_library(application_ui src/ui/books_list.cpp src/ui/reader_slice.cpp)
target_link_libraries(application_ui PUBLIC application_core)
target_compile_definitions(application_ui PRIVATE EPUBREADER_VERSION="${PROJECT_VERSION}")
target_compile_options(application_ui PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(epubreader src/app/reader_main.cpp)
target_link_libraries(epubreader PRIVATE application_ui)
target_compile_options(epubreader PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_compile_definitions(epubreader PRIVATE EPUBREADER_VERSION="${PROJECT_VERSION}")
if(EPUBREADER_DEVICE)
    target_compile_definitions(epubreader-diagnostic PRIVATE EPUBREADER_DEVICE=1)
    target_compile_definitions(epubreader PRIVATE EPUBREADER_DEVICE=1)
    # Find bundled shared libraries even when launched without the shell wrapper.
    set_target_properties(epubreader-diagnostic PROPERTIES BUILD_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN/lib")
    set_target_properties(epubreader PROPERTIES BUILD_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN/lib")
endif()
if(EPUBREADER_SANITIZE)
    target_compile_options(diagnostic_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(diagnostic_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(power-tests tests/power_tests.cpp)
    target_link_libraries(power-tests PRIVATE application_core)
    add_test(NAME power-core COMMAND power-tests "${CMAKE_BINARY_DIR}/power-test-state")
    set_tests_properties(power-core PROPERTIES TIMEOUT 20)
    add_executable(music-tests tests/music_tests.cpp)
    target_link_libraries(music-tests PRIVATE diagnostic_core)
    add_test(NAME music-core COMMAND music-tests "${CMAKE_SOURCE_DIR}/tests/fixtures" "${CMAKE_BINARY_DIR}/music-test-state")
    set_tests_properties(music-core PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy" TIMEOUT 30)
    add_executable(storage-tests tests/storage_tests.cpp)
    target_link_libraries(storage-tests PRIVATE production_storage)
    add_test(NAME production-storage COMMAND storage-tests "${CMAKE_BINARY_DIR}/storage-test-state")
    add_executable(library-tests tests/library_tests.cpp)
    target_link_libraries(library-tests PRIVATE library_core)
    add_test(NAME library-core COMMAND library-tests "${CMAKE_BINARY_DIR}/library-test-state" "${CMAKE_SOURCE_DIR}/assets/books/prose-epub2.epub")
    add_executable(controller-tests tests/controller_tests.cpp)
    target_link_libraries(controller-tests PRIVATE application_ui)
    add_test(NAME application-core COMMAND controller-tests "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_SOURCE_DIR}/tests/fixtures" "${CMAKE_BINARY_DIR}/application-test-state" "$<TARGET_FILE:epubreader>" "${CMAKE_BINARY_DIR}/library-screenshots")

    target_link_libraries(page-turn-tests PRIVATE application_ui)
    foreach(version 2 3)
        add_test(NAME reader-page-turns-epub${version} COMMAND page-turn-tests

        set_tests_properties(reader-page-turns-epub${version} PROPERTIES TIMEOUT 120)
    endforeach()
    add_test(NAME reader-page-turns-blank COMMAND page-turn-tests
        "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_SOURCE_DIR}/tests/fixtures/blank-pages.epub"
        "${CMAKE_BINARY_DIR}/page-turns-blank" "$<TARGET_FILE:epubreader>" --expect-blank)
    set_tests_properties(reader-page-turns-blank PROPERTIES TIMEOUT 120)

        add_test(NAME reader-page-turns-reported COMMAND page-turn-tests

        "$<TARGET_FILE:epubreader>" "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_SOURCE_DIR}/tests/fixtures"
        "${CMAKE_BINARY_DIR}/reader-smoke-state" "${CMAKE_BINARY_DIR}/reader-screenshots")

    add_executable(diagnostic-tests tests/diagnostic_tests.cpp)
    target_link_libraries(diagnostic-tests PRIVATE diagnostic_core)
    add_test(NAME diagnostic-core COMMAND diagnostic-tests "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/test-state")
    set_tests_properties(diagnostic-core PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
    add_test(NAME diagnostic-smoke COMMAND epubreader-diagnostic --assets "${CMAKE_SOURCE_DIR}/assets" --state-dir "${CMAKE_BINARY_DIR}/smoke-state" --smoke-test "${CMAKE_BINARY_DIR}/screenshots")
    set_tests_properties(diagnostic-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
