cmake_minimum_required(VERSION 3.20)
project(epubreader VERSION 0.2.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EPUBREADER_DEVICE "Build for OpenDingux" OFF)
option(EPUBREADER_SANITIZE "Host address/undefined sanitizers" OFF)
add_compile_options("-ffile-prefix-map=${PROJECT_SOURCE_DIR}=.")
if(EPUBREADER_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(PLATFORM REQUIRED IMPORTED_TARGET sdl SDL_ttf libmpg123 alsa)
find_package(Threads REQUIRED)
include(cmake/CRengine.cmake)
add_library(diagnostic_core
    src/ui/canvas.cpp src/platform/input.cpp src/platform/events.cpp src/storage/state.cpp
    src/platform/mixer.cpp src/audio/audio.cpp)
target_include_directories(diagnostic_core PUBLIC src)
target_link_libraries(diagnostic_core PUBLIC PkgConfig::PLATFORM Threads::Threads)
target_compile_options(diagnostic_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(reader_core src/reader/engine.cpp src/reader/worker.cpp src/storage/position.cpp src/ui/reader_controls.cpp)
target_link_libraries(reader_core PUBLIC crengine diagnostic_core)
target_compile_options(reader_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(epubreader-diagnostic src/app/main.cpp src/ui/diagnostic.cpp)
target_link_libraries(epubreader-diagnostic PRIVATE diagnostic_core)
target_compile_options(epubreader-diagnostic PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_compile_definitions(epubreader-diagnostic PRIVATE EPUBREADER_VERSION="${PROJECT_VERSION}")
add_executable(epubreader src/app/reader_main.cpp src/ui/reader_slice.cpp)
target_link_libraries(epubreader PRIVATE reader_core)
target_compile_options(epubreader PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_compile_definitions(epubreader PRIVATE EPUBREADER_VERSION="${PROJECT_VERSION}")
if(EPUBREADER_DEVICE)
    target_compile_definitions(epubreader-diagnostic PRIVATE EPUBREADER_DEVICE=1)
    target_compile_definitions(epubreader PRIVATE EPUBREADER_DEVICE=1)
    # Find bundled shared libraries even when launched without the shell wrapper.
    set_target_properties(epubreader-diagnostic PROPERTIES BUILD_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN/lib")
    set_target_properties(epubreader PROPERTIES BUILD_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN/lib")
endif()
if(EPUBREADER_SANITIZE)
    target_compile_options(diagnostic_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(diagnostic_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(reader-tests tests/reader_tests.cpp)
    target_link_libraries(reader-tests PRIVATE reader_core)
    add_test(NAME reader-core COMMAND reader-tests "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_SOURCE_DIR}/tests/fixtures" "${CMAKE_BINARY_DIR}/reader-test-state")
    set_tests_properties(reader-core PROPERTIES TIMEOUT 120)
    add_test(NAME reader-smoke COMMAND epubreader --assets "${CMAKE_SOURCE_DIR}/assets" --book "${CMAKE_SOURCE_DIR}/assets/books/prose-epub3.epub" --state-dir "${CMAKE_BINARY_DIR}/reader-smoke-state" --smoke-test "${CMAKE_BINARY_DIR}/reader-screenshots")
    set_tests_properties(reader-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 60)
    add_executable(diagnostic-tests tests/diagnostic_tests.cpp)
    target_link_libraries(diagnostic-tests PRIVATE diagnostic_core)
    add_test(NAME diagnostic-core COMMAND diagnostic-tests "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/test-state")
    set_tests_properties(diagnostic-core PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
    add_test(NAME diagnostic-smoke COMMAND epubreader-diagnostic --assets "${CMAKE_SOURCE_DIR}/assets" --state-dir "${CMAKE_BINARY_DIR}/smoke-state" --smoke-test "${CMAKE_BINARY_DIR}/screenshots")
    set_tests_properties(diagnostic-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
